const SENHA_ADMIN = "7741"; // DEFINA SUA SENHA AQUI

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // Mantém sua lógica original de salvar novos comentários
    sheet.appendRow([new Date(), data.nome, data.email, data.comentario, "PENDENTE"]);
    
    return ContentService.createTextOutput(JSON.stringify({"result":"success"}))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch (f) {
    return ContentService.createTextOutput(JSON.stringify({"result":"error", "error": f.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();

  // --- NOVA LÓGICA: APROVAR COMENTÁRIO ---
  if (e.parameter.action === 'aprovar' && e.parameter.pw === SENHA_ADMIN) {
    var linhaParaAprovar = parseInt(e.parameter.row);
    sheet.getRange(linhaParaAprovar, 5).setValue("SIM");
    return ContentService.createTextOutput("Sucesso");
  }

  // --- NOVA LÓGICA: EXCLUIR COMENTÁRIO ---
  if (e.parameter.action === 'excluir' && e.parameter.pw === SENHA_ADMIN) {
    var linhaParaExcluir = parseInt(e.parameter.row);
    sheet.deleteRow(linhaParaExcluir);
    return ContentService.createTextOutput("Excluído");
  }

  // --- NOVA LÓGICA: LISTAR PARA O PAINEL ADMIN (TUDO) ---
  if (e.parameter.mode === 'admin' && e.parameter.pw === SENHA_ADMIN) {
    var allData = [];
    for (var i = 1; i < rows.length; i++) {
      allData.push({
        row: i + 1,
        nome: rows[i][1],
        email: rows[i][2],
        comentario: rows[i][3],
        status: rows[i][4]
      });
    }
    return ContentService.createTextOutput(JSON.stringify(allData))
                         .setMimeType(ContentService.MimeType.JSON);
  }

  // --- SUA LÓGICA ORIGINAL: LISTAR NO SITE (SÓ "SIM") ---
  var dataPublica = [];
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][4] === "SIM") { 
      dataPublica.push({
        data: rows[i][0],
        nome: rows[i][1],
        comentario: rows[i][3]
      });
    }
  }

  return ContentService.createTextOutput(JSON.stringify(dataPublica))
                       .setMimeType(ContentService.MimeType.JSON);
}

// MANTÉM SUA TRAVA DE SEGURANÇA ORIGINAL
function onEdit(e) {
  var range = e.range;
  var coluna = range.getColumn();
  var linha = range.getRow();

  if (linha === 1 || coluna === 5) return;

  if (coluna >= 1 && coluna <= 4) {
    var valorAntigo = e.oldValue;
    if (valorAntigo !== undefined) {
      range.setValue(valorAntigo);
      SpreadsheetApp.getUi().alert("Atenção: Os dados originais do leitor não devem ser alterados. Use a coluna Status ou exclua a linha.");
    }
  }
}
